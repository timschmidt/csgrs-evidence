use hyperbrep::{builder,boolean,Point3,Real};
fn p(x:i32,y:i32,z:i32)->Point3 {Point3::new(x.into(),y.into(),z.into())}
fn main() {
 let torus=builder::torus(Real::from(5),Real::from(2)).unwrap();
 let half=builder::cuboid(p(-20,-20,0),p(20,20,20)).unwrap();
 let longitudinal=builder::cuboid(p(0,-20,-20),p(20,20,20)).unwrap();
 let cone=builder::cone_frustum(5.into(),2.into(),6.into()).unwrap();
 for (name,result) in [("torus axial half",boolean::intersection(&torus,&half)),("torus longitudinal half",boolean::intersection(&torus,&longitudinal)),("cone longitudinal half",boolean::intersection(&cone,&longitudinal))] { match result { Ok(model)=>println!("{name}: success bodies={}",model.bodies().len()), Err(error)=>println!("{name}: {error:?}") } }
}

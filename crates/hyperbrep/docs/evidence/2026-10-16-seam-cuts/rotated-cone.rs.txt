#[allow(dead_code)]
#[path="/mnt/development/csgrs/crates/hyperbrep/tests/support/seam_cuts.rs"] mod acceptance;
use hyperbrep::{boolean,FacePairRelation,FacePairTrim};
fn main(){
 let (source,cutter,_,_)=acceptance::fixture(2);
 let map=acceptance::similarity(&acceptance::ratio(1,1000),false);
 let source=source.transformed(&map).unwrap();let cutter=cutter.transformed(&map).unwrap();
 let graph=boolean::intersection_graph(source.bodies().next().unwrap(),cutter.bodies().next().unwrap()).unwrap();
 println!("unsupported={} unresolved={} traces={}",graph.unsupported_pairs(),graph.unresolved_trim_pairs(),graph.trimmed_curve_fragments());
 for pair in graph.intersections(){
  let f=pair.first_face();let s=pair.second_face();
  let kind=match pair.relation(){FacePairRelation::Unsupported=>"unsupported",_=>"exact"};
  let trim=match pair.trim(){FacePairTrim::NotAvailable=>"unavailable",FacePairTrim::SurfaceCurveFragments(_)=>"retained",_=>"other"};
  println!("first={} {:?} second={} {:?} {kind} {trim}",f.index(),f.surface().kind(),s.index(),s.surface().kind());
 }
}

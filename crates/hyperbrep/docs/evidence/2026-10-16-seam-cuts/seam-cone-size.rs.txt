use hyperbrep::{Real, boolean};
#[allow(dead_code)]
#[path="/mnt/development/csgrs/crates/hyperbrep/tests/support/seam_cuts.rs"] mod acceptance;
fn main(){
 let (source,cutter,_,_)=acceptance::fixture(2);
 let map=acceptance::similarity(&Real::from(2).sqrt().unwrap(),false);
 let source=source.transformed(&map).unwrap(); let cutter=cutter.transformed(&map).unwrap();
 for (name,model) in [("source",source.model().clone()),("cutter",cutter.model().clone()),("intersection",boolean::intersection(&source,&cutter).unwrap().into_model()),("swapped",boolean::intersection(&cutter,&source).unwrap().into_model()),("difference",boolean::difference(&source,&cutter).unwrap().into_model())] {
  let json=model.to_json().unwrap(); println!("{name} bytes={}",json.len());
  std::fs::write(format!("/tmp/seam-cone-size-{name}.json"),json).unwrap();
 }
}

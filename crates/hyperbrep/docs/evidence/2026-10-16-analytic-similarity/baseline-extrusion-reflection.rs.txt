use hyperbrep::{builder, ArchiveLimits, Curve3, Matrix4, Point3, RawModel, Real, Vector3};
fn main() {
    let profile=Curve3::line(Point3::new(2.into(),0.into(),0.into()),Point3::new(4.into(),0.into(),2.into())).unwrap();
    let source=builder::extrusion_patch(profile,Vector3::y(),0.into(),3.into()).unwrap();
    let reflected=source.model().transformed(&Matrix4::affine_nonuniform_scale([-Real::one(),Real::one(),Real::one()])).unwrap();
    let json=reflected.to_json().unwrap();
    let replay=RawModel::from_json(&json,&ArchiveLimits::default()).unwrap().validate();
    println!("reflected_extrusion_replay={:?}",replay.as_ref().map(|model| model.counts()));
    assert!(replay.is_err(),"the retained baseline unexpectedly admitted the invalid reflection");
}

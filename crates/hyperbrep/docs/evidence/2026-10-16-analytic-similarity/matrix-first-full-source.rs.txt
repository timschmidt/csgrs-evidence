//! Exact expectations shared by native and WASM similarity consumers.
use hyperbrep::{
    ArchiveLimits, BodySet, Matrix4, Model, Point2, Point3, RawModel, Real, SolidPointLocation,
    builder,
};
use hyperlimit::{compare_reals, point3_equal};

pub fn ratio(n: i32, d: i32) -> Real {
    (Real::from(n) / Real::from(d)).unwrap()
}

pub fn equal(actual: &Real, expected: &Real) {
    assert_eq!(
        compare_reals(actual, expected, hyperbrep::STRICT_PREDICATES).value(),
        Some(std::cmp::Ordering::Equal),
        "actual={actual:?}, expected={expected:?}"
    );
}

pub fn point(x: i32, y: i32, z: i32) -> Point3 {
    Point3::new(x.into(), y.into(), z.into())
}

pub fn similarity(scale: &Real, reflected: bool) -> Matrix4 {
    let sign = if reflected { -Real::one() } else { Real::one() };
    Matrix4::from_row_major([
        scale * ratio(3, 5) * &sign,
        -scale * ratio(4, 5),
        0.into(),
        3.into(),
        scale * ratio(4, 5) * sign,
        scale * ratio(3, 5),
        0.into(),
        (-2).into(),
        0.into(),
        0.into(),
        scale.clone(),
        7.into(),
        0.into(),
        0.into(),
        0.into(),
        1.into(),
    ])
}

pub fn reload(model: &Model) -> Model {
    let json = model.to_json().unwrap();
    let loaded = RawModel::from_json(&json, &ArchiveLimits::default())
        .unwrap()
        .validate()
        .unwrap();
    assert_eq!(loaded.counts(), model.counts());
    assert_eq!(loaded.to_json().unwrap(), json);
    loaded
}

pub fn check_images(model: &Model) {
    for edge_use in model.edge_uses() {
        let pcurve = edge_use.pcurve();
        for parameter in [pcurve.domain_start().unwrap(), pcurve.domain_end().unwrap()] {
            let uv = pcurve.point_at(&parameter).unwrap();
            let Some(uv) = uv.coordinates() else { continue };
            let on_surface = edge_use
                .wire()
                .face()
                .surface()
                .point_at(&Point2::new(uv.x().clone(), uv.y().clone()))
                .unwrap();
            let edge_parameter = edge_use.edge_parameter_at(&parameter).unwrap();
            let on_edge = edge_use.edge().curve().point_at(&edge_parameter).unwrap();
            assert!(on_edge.coincides_with(&on_surface.into()).unwrap());
        }
    }
}

pub fn primitives() {
    let cases: Vec<(&str, BodySet, Real, Real, Point3, Point3)> = vec![
        (
            "cylinder",
            builder::cylinder(5.into(), 7.into()).unwrap(),
            Real::from(175) * Real::pi(),
            Real::from(120) * Real::pi(),
            point(0, 0, 3),
            point(5, 0, 3),
        ),
        (
            "sphere",
            builder::sphere(5.into()).unwrap(),
            ratio(500, 3) * Real::pi(),
            Real::from(100) * Real::pi(),
            point(0, 0, 0),
            point(5, 0, 0),
        ),
        (
            "cone",
            builder::cone_frustum(5.into(), 2.into(), 6.into()).unwrap(),
            Real::from(78) * Real::pi(),
            (Real::from(29) + Real::from(21) * Real::from(5).sqrt().unwrap()) * Real::pi(),
            point(0, 0, 3),
            point(5, 0, 0),
        ),
        (
            "torus",
            builder::torus(5.into(), 2.into()).unwrap(),
            Real::from(40) * Real::pi() * Real::pi(),
            Real::from(40) * Real::pi() * Real::pi(),
            point(5, 0, 0),
            point(7, 0, 0),
        ),
        (
            "revolution",
            builder::revolve(&[
                Point2::new(2.into(), 0.into()),
                Point2::new(4.into(), 0.into()),
                Point2::new(4.into(), 3.into()),
                Point2::new(2.into(), 3.into()),
            ])
            .unwrap(),
            Real::from(36) * Real::pi(),
            Real::from(60) * Real::pi(),
            point(3, 0, 1),
            point(4, 0, 1),
        ),
    ];
    for (name, source, volume, area, inside, boundary) in cases {
        for scale in [
            ratio(1, 1000),
            ratio(127, 5000),
            Real::from(2),
            Real::from(2).sqrt().unwrap(),
        ] {
            for reflected in [false, true] {
                let transform = similarity(&scale, reflected);
                let moved = source
                    .transformed(&transform)
                    .unwrap_or_else(|e| panic!("{name}, {scale:?}, reflected={reflected}: {e:?}"));
                let loaded = reload(moved.model());
                for model in [moved.model(), &loaded] {
                    let body = model.solids().next().unwrap();
                    equal(
                        &body.volume().unwrap(),
                        &(&volume * &scale * &scale * &scale),
                    );
                    equal(
                        &model
                            .faces()
                            .map(|f| f.area().unwrap())
                            .fold(Real::zero(), |sum, a| sum + a),
                        &(&area * &scale * &scale),
                    );
                    for (p, expected) in [
                        (&inside, SolidPointLocation::Inside),
                        (&boundary, SolidPointLocation::Boundary),
                        (&point(20, 20, 20), SolidPointLocation::Outside),
                    ] {
                        assert_eq!(
                            body.classify_point(&transform.transform_point3(p).unwrap())
                                .unwrap(),
                            expected,
                            "{name}"
                        );
                    }
                    check_images(model);
                }
                let inverse = loaded.transformed(&transform.inverse().unwrap()).unwrap();
                let restored = reload(&inverse);
                equal(
                    &restored.solids().next().unwrap().volume().unwrap(),
                    &volume,
                );
                for (before, after) in source.model().vertices().zip(restored.vertices()) {
                    assert_eq!(
                        point3_equal(
                            before.point().try_coordinates().unwrap(),
                            after.point().try_coordinates().unwrap(),
                            hyperbrep::STRICT_PREDICATES
                        )
                        .value(),
                        Some(true)
                    );
                }
            }
        }
    }
}

pub fn clipped_and_voids() {
    use hyperbrep::{builder::SphereVoid, boolean};
    let sphere = builder::sphere(5.into()).unwrap();
    let second = sphere.transformed(&Matrix4::affine_translation([0.into(),0.into(),6.into()])).unwrap();
    let cylinder = builder::cylinder(3.into(),12.into()).unwrap()
        .transformed(&Matrix4::affine_translation([0.into(),0.into(),(-6).into()])).unwrap();
    let inner_cylinder = builder::cylinder(1.into(),2.into()).unwrap()
        .transformed(&Matrix4::affine_translation([0.into(),0.into(),(-1).into()])).unwrap();
    let half = builder::cuboid(point(-20,-20,0),point(20,20,20)).unwrap();
    let longitudinal = builder::cuboid(point(0,-20,-20),point(20,20,20)).unwrap();
    let slab = builder::cuboid(point(-20,-20,-3),point(20,20,3)).unwrap();
    let torus = builder::torus(5.into(),2.into()).unwrap();
    let cone = builder::cone_frustum(5.into(),2.into(),6.into()).unwrap();
    let profile = |points: &[(i32,i32)]| points.iter().map(|&(x,y)| Point2::new(x.into(),y.into())).collect::<Vec<_>>();
    let cases = vec![
        ("spherical void", builder::sphere_with_voids(5.into(), &[SphereVoid { center: point(1,0,0), radius: 1.into() }]).unwrap(), ratio(496,3)*Real::pi()),
        ("sphere union", boolean::union(&sphere,&second).unwrap(), ratio(896,3)*Real::pi()),
        ("sphere lens", boolean::intersection(&sphere,&second).unwrap(), ratio(104,3)*Real::pi()),
        ("sphere difference", boolean::difference(&sphere,&second).unwrap(), Real::from(132)*Real::pi()),
        ("spherical cap", boolean::intersection(&sphere,&half).unwrap(), ratio(250,3)*Real::pi()),
        ("spherical band", boolean::intersection(&sphere,&slab).unwrap(), Real::from(132)*Real::pi()),
        ("radial intersection", boolean::intersection(&sphere,&cylinder).unwrap(), ratio(244,3)*Real::pi()),
        ("radial difference", boolean::difference(&sphere,&cylinder).unwrap(), ratio(256,3)*Real::pi()),
        ("finite union", boolean::union(&sphere,&cylinder).unwrap(), ratio(580,3)*Real::pi()),
        ("two components", boolean::difference(&cylinder,&sphere).unwrap(), ratio(80,3)*Real::pi()),
        ("cylindrical void", boolean::difference(&sphere,&inner_cylinder).unwrap(), ratio(494,3)*Real::pi()),
        ("sphere inside cylinder", boolean::difference(&cylinder,&builder::sphere(1.into()).unwrap()).unwrap(), ratio(320,3)*Real::pi()),
        ("torus axial half", boolean::intersection(&torus,&half).unwrap(), Real::from(20)*Real::pi()*Real::pi()),
        ("torus longitudinal half", boolean::intersection(&torus,&longitudinal).unwrap(), Real::from(20)*Real::pi()*Real::pi()),
        ("cone half", boolean::intersection(&cone,&longitudinal).unwrap(), Real::from(39)*Real::pi()),
        ("revolution cavity", builder::revolve_region(&profile(&[(1,0),(4,0),(4,4),(1,4)]), &[profile(&[(2,1),(3,1),(3,2),(2,2)])]).unwrap(), Real::from(55)*Real::pi()),
    ];
    let volume = |model: &Model| model.solids().map(|b|b.volume().unwrap()).fold(Real::zero(),|s,v|s+v);
    for (name,source,expected) in cases {
        equal(&volume(source.model()), &expected);
        for scale in [ratio(127,5000),Real::from(2).sqrt().unwrap()] {
            for reflected in [false,true] {
                let transform = similarity(&scale,reflected);
                let moved = source.transformed(&transform).unwrap_or_else(|e| panic!("{name}: {e:?}"));
                let loaded = reload(moved.model());
                equal(&volume(&loaded), &(&expected * &scale * &scale * &scale));
                for (before,after) in source.model().solids().zip(loaded.solids()) {
                    for point in [point(0,0,0),point(0,0,4),point(2,0,1),point(4,0,1),point(8,0,0),point(0,0,-5)] {
                        assert_eq!(after.classify_point(&transform.transform_point3(&point).unwrap()).unwrap(),before.classify_point(&point).unwrap(),"{name}");
                    }
                }
                check_images(&loaded);
            }
        }
    }
}

pub fn selected_and_repeated() {
    use hyperbrep::{Curve3,Curve3Kind,CurveParameterLocation,Parameter,boolean};
    let selector = Curve3::rational_bezier([-5,-7,-9,-11,-13,145]
        .map(|n|Point3::new(ratio(n,5),0.into(),0.into())).to_vec(),vec![Real::one();6]).unwrap();
    let CurveParameterLocation::Parameters(parameters) = selector.parameters_of(&Point3::origin()).unwrap() else { panic!("selected root") };
    let root: Parameter = parameters[0].clone();
    assert!(root.try_real().is_none());
    let source=builder::cylinder(5.into(),7.into()).unwrap();
    let split=source.model().edges().find(|e|e.curve().kind()==Curve3Kind::Line).unwrap().split(root.clone()).unwrap();
    let mut current=split.model().clone();
    let map=similarity(&ratio(127,5000),true);
    let inverse=map.clone().inverse().unwrap();
    for _ in 0..3 {
        let moved=reload(&current.transformed(&map).unwrap());
        for (before,after) in current.edges().zip(moved.edges()) {
            assert_eq!(before.domain().start().compare(after.domain().start()).unwrap(),std::cmp::Ordering::Equal);
            assert_eq!(before.domain().end().compare(after.domain().end()).unwrap(),std::cmp::Ordering::Equal);
        }
        assert_eq!(current.vertices().filter(|v|v.point().try_coordinates().is_none()).count(),moved.vertices().filter(|v|v.point().try_coordinates().is_none()).count());
        let continued=moved.edges().find(|e|e.domain().end().try_real().is_none()).unwrap().split(ratio(1,2).into()).unwrap();
        current=reload(&continued.model().transformed(&inverse).unwrap());
        equal(&current.solids().next().unwrap().volume().unwrap(), &(Real::from(175)*Real::pi()));
    }
    let result=BodySet::from_solid(current.solids().next().unwrap());
    equal(&boolean::intersection(&result,&source).unwrap().bodies().next().unwrap().volume().unwrap(), &(Real::from(175)*Real::pi()));
    assert!(boolean::difference(&result,&source).unwrap().is_empty());
    let empty=BodySet::empty().transformed(&map).unwrap();
    assert!(empty.is_empty());
    assert_eq!(boolean::union(&empty,&result).unwrap().bodies().count(),1);
}

pub fn rejection_and_recovery() {
    use hyperbrep::GeometryError;
    let source=builder::cylinder(5.into(),7.into()).unwrap();
    let json=source.model().to_json().unwrap();
    let nonuniform=Matrix4::affine_nonuniform_scale([2.into(),3.into(),4.into()]);
    assert!(matches!(source.transformed(&nonuniform),Err(GeometryError::UnsupportedTransform)));
    assert!(matches!(source.transformed(&Matrix4::affine_nonuniform_scale([1.into(),0.into(),1.into()])),Err(GeometryError::SingularTransform)));
    let mut perspective=Matrix4::identity(); perspective.0[3][0]=ratio(1,1000);
    assert!(matches!(source.transformed(&perspective),Err(GeometryError::NonAffineTransform)));
    assert_eq!(source.model().to_json().unwrap(),json);
    let moved=source.transformed(&Matrix4::affine_nonuniform_scale([(-2).into(),(-2).into(),(-2).into()])).unwrap();
    equal(&reload(moved.model()).solids().next().unwrap().volume().unwrap(), &(Real::from(1400)*Real::pi()));
    let box_model=builder::cuboid(point(0,0,0),point(2,3,4)).unwrap().transformed(&nonuniform).unwrap();
    equal(&reload(box_model.model()).solids().next().unwrap().volume().unwrap(),&Real::from(576));
}

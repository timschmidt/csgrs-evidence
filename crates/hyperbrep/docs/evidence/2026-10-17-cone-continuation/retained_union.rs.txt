use hyperbrep::{ArchiveLimits, BodySet, RawModel, boolean};
fn main() {
    let operands: Vec<_> = std::env::args().skip(1).map(|path| {
        let json = std::fs::read_to_string(path).unwrap();
        let model = RawModel::from_json(&json, &ArchiveLimits::default()).unwrap().validate().unwrap();
        BodySet::from_solid(model.solids().next().unwrap())
    }).collect();
    let result = boolean::union(&operands[0], &operands[1]).unwrap();
    println!("UNION_OK {}", result.model().to_json().unwrap().len());
}
